type nativeOwnerKey struct{}

type NativePoolOwner struct {
 ctx context.Context
 cancel context.CancelFunc
 mu sync.Mutex
 stopping bool
 active int
 bodies map[*nativeBody]struct{}
 clients map[*http.Client]*nativeClient
 done chan struct{}
 finish sync.Once
}

func NewNativePoolOwner() *NativePoolOwner {
 ctx, cancel := context.WithCancel(context.Background())
 return &NativePoolOwner{ctx:ctx,cancel:cancel,bodies:make(map[*nativeBody]struct{}),clients:make(map[*http.Client]*nativeClient),done:make(chan struct{})}
}

func (o *NativePoolOwner) Context() context.Context { return context.WithValue(o.ctx,nativeOwnerKey{},o) }

func nativeOwnerFrom(ctx context.Context) *NativePoolOwner {
 owner, _ := ctx.Value(nativeOwnerKey{}).(*NativePoolOwner)
 return owner
}

func nativeHTTPClient(ctx context.Context,destination net.Destination,settings *internet.MemoryStreamConfig) DialerClient {
 client:=createHTTPClient(destination,settings)
 if owner:=nativeOwnerFrom(ctx);owner!=nil {
  if concrete,ok:=client.(*DefaultDialerClient);ok {
   concrete.nativeOwner=owner
   owner.mu.Lock()
   owner.clients[concrete.client]=&nativeClient{dialer:concrete}
   owner.mu.Unlock()
  }
 }
 return client
}

func nativeForgetClient(client *DefaultDialerClient) {
 if owner:=client.nativeOwner;owner!=nil {
  owner.mu.Lock()
  if current:=owner.clients[client.client];current!=nil {
   current.retired=true
   if current.active==0 {
    client.client.CloseIdleConnections()
    delete(owner.clients,client.client)
   }
  }
  owner.mu.Unlock()
 }
}

type nativeClient struct { dialer *DefaultDialerClient; active int; retired bool }

func (o *NativePoolOwner) closed() bool {
 o.mu.Lock()
 defer o.mu.Unlock()
 return o.stopping
}

type nativeContext struct { context.Context; life context.Context }
func (c nativeContext) Done() <-chan struct{} { return c.life.Done() }
func (c nativeContext) Err() error { return c.life.Err() }
func (c nativeContext) Deadline() (time.Time,bool) { return c.life.Deadline() }

func nativeRequestContext(ctx context.Context) context.Context {
 base := context.WithoutCancel(ctx)
 if owner := nativeOwnerFrom(ctx); owner != nil { return nativeContext{base,owner.ctx} }
 return base
}

func nativeDo(client *http.Client, request *http.Request) (*http.Response,error) {
 owner := nativeOwnerFrom(request.Context())
 if owner == nil { return client.Do(request) }
 owner.mu.Lock()
 current:=owner.clients[client]
 if owner.stopping || current==nil { owner.mu.Unlock(); return nil,context.Canceled }
 owner.active++
 current.active++
 owner.mu.Unlock()
 response,err := client.Do(request)
 if err != nil { owner.release(nil,client); return nil,err }
 body := &nativeBody{ReadCloser:response.Body,owner:owner,client:client}
 owner.mu.Lock()
 if owner.stopping {
  owner.mu.Unlock()
  body.Close()
  return nil,context.Canceled
 }
 owner.bodies[body]=struct{}{}
 owner.mu.Unlock()
 response.Body=body
 return response,nil
}

type nativeBody struct { io.ReadCloser; owner *NativePoolOwner; client *http.Client; once sync.Once }
func (b *nativeBody) Read(data []byte) (int,error) {
 count,err := b.ReadCloser.Read(data)
 if err != nil { b.Close() }
 return count,err
}
func (b *nativeBody) Close() error {
 var err error
 b.once.Do(func(){ err=b.ReadCloser.Close(); b.owner.release(b,b.client) })
 return err
}

func (o *NativePoolOwner) release(body *nativeBody,client *http.Client) {
 o.mu.Lock()
 defer o.mu.Unlock()
 delete(o.bodies,body)
 if current:=o.clients[client];current!=nil {
  current.active--
  if current.retired && current.active==0 {
   client.CloseIdleConnections()
   delete(o.clients,client)
  }
 }
 o.active--
 if o.stopping && o.active==0 { o.finish.Do(func(){close(o.done)}) }
}

func (o *NativePoolOwner) BeginClose() {
 o.mu.Lock()
 o.stopping=true
 o.cancel()
 bodies:=make([]*nativeBody,0,len(o.bodies))
 for body := range o.bodies { bodies=append(bodies,body) }
 if o.active==0 { o.finish.Do(func(){close(o.done)}) }
 o.mu.Unlock()
 for _,body:=range bodies { body.Close() }
}

func (o *NativePoolOwner) Close(ctx context.Context) error {
 o.BeginClose()
 select { case <-o.done: case <-ctx.Done(): return ctx.Err() }
 globalDialerAccess.Lock()
 defer globalDialerAccess.Unlock()
 for key,manager:=range globalDialerMap {
  if key.owner!=o { continue }
  delete(globalDialerMap,key)
  for _,connection:=range manager.xmuxClients {
   connection.NotUsed.Store(true)
  }
 }
 o.mu.Lock()
 clients:=make([]*DefaultDialerClient,0,len(o.clients))
 for client,current:=range o.clients {
  clients=append(clients,current.dialer)
  delete(o.clients,client)
 }
 o.mu.Unlock()
 for _,client:=range clients { client.closed.Store(true); client.Close() }
 return nil
}

func (o *NativePoolOwner) PoolCount() int {
 globalDialerAccess.Lock()
 defer globalDialerAccess.Unlock()
 count:=0
 for key:=range globalDialerMap { if key.owner==o { count++ } }
 return count
}

type nativeClosedClient struct{}
func (nativeClosedClient) IsClosed() bool { return true }
func (nativeClosedClient) OpenStream(context.Context,string,string,io.Reader,bool)(io.ReadCloser,net.Addr,net.Addr,error){ return nil,nil,nil,context.Canceled }
func (nativeClosedClient) PostPacket(context.Context,string,string,string,buf.MultiBuffer)error { return context.Canceled }
